def query(question, model):
    """Query the model for intent and retrieve star data."""
    q = question.lower()

    # Predict intent
    seq = tokenizer.texts_to_sequences([q])
    seq = pad_sequences(seq, maxlen=15)
    intent_id = np.argmax(model.predict(seq, verbose=0))
    intent = le.inverse_transform([intent_id])[0]
    
    star = None
    for name in df["Name_normalized"]:
        if name in q:
            star = name
            break
    
    if star is None:
        return "no star found"
    
    row = df[df["Name_normalized"] == star].iloc[0]

    if intent == "exists":
        return f"Yes, {row['Name']} exists."
    elif intent == "vmag":
        return f"The Vmag of {row['Name']} is {row['Vmag']}."
    elif intent == "color":
        return f"The B-V value of {row['Name']} is {row['B-V']}."
    elif intent == "spectral":
        return f"The spectral type of {row['Name']} is {row['SpType']}."
    elif intent == "parallax":
        return f"The parallax of {row['Name']} is {row['Parallax']}."
